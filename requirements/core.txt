# Minimal runtime set: the core, i.e. everything in the contract_uav_core
# package except contract_uav_core/planning/* (needs pacti, see
# requirements-legacy.txt) and contract_uav_core/viz/* (needs pygame, see
# requirements-viz.txt). contract_uav_core/setup.py declares the same pins.
# The package itself is installed separately:
#   python -m pip install -e ./contract_uav_core
# Example: python -m contract_uav_core.core
#
# Python floor: 3.7
#   - numpy 1.20.0, the lowest version allowed below, declares
#     Requires-Python >=3.7 on PyPI.
#   - The code itself needs 3.7 (dataclasses; static check with `vermin`).
#   Python below 3.10 has not been exercised; CI covers 3.10 and 3.11.
numpy>=1.20.0
