\warningbox{
\textbf{NED Altitude Convention:} In the NED frame:
\begin{itemize}
    \item Altitude above ground = $-z$ (e.g., 10m altitude $\Rightarrow z = -10$)
    \item Ascending = decreasing $z$ (negative $\dot{z}$)
    \item Descending = increasing $z$ (positive $\dot{z}$)
\end{itemize}
