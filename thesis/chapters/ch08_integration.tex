\section{Main Control Loop}

The control loop executes at 50 Hz (20ms period):

\begin{algorithm}[H]
\caption{Main Control Loop}
\begin{algorithmic}[1]
    \State \textbf{Input:} Raw sensor data, mission commands
    \State \textbf{Output:} Safe actuator commands
    \Procedure{ControlStep}{sensors, time}
        \State \Comment{1. Sensor Processing}
        \State measurements, raw\_sensors $\gets$ process\_sensors(sensors)
        \State
        \State \Comment{2. State Estimation (PREDICT then UPDATE)}
        \State ekf.predict(dt)
        \State state, estimation\_ok $\gets$ ekf.update(measurements, raw\_sensors, time)
        \State
        \State \Comment{3. Condition Estimation}
        \State conditions $\gets$ estimate\_conditions(state, raw\_sensors)
        \State gps\_ok $\gets$ ekf.check\_sensor\_contracts()['gps\_valid']
        \State
        \State \Comment{4. Horizon Planning (at 2 Hz)}
        \State controller\_rec, margin $\gets$ planner.update(state, conditions, time)
        \State
        \State \Comment{5. Supervision --- Mode and Setpoint}
        \State setpoint, ctrl\_name $\gets$ supervisor.update(state, conditions, gps\_ok, time)
        \State
        \State \Comment{6. Controller Switching}
        \If{ctrl\_name $\neq$ current\_controller}
            \State controller\_switcher.switch\_to(ctrl\_name)
        \EndIf
        \State
        \State \Comment{7. Control Computation}
        \State nominal\_control $\gets$ controller\_switcher.compute(state, setpoint)
        \State
        \State \Comment{8. Safety Filtering}
        \State safe\_control, intervened $\gets$ cbf\_filter.filter(state, nominal\_control, time)
        \State
        \State \Comment{9. Logging and Monitoring}
        \State log\_telemetry(state, control, conditions, time)
        \State
        \State \Return safe\_control
    \EndProcedure
\end{algorithmic}
\end{algorithm}

\section{Condition Estimation}

The system estimates environmental and tracking conditions:

\begin{lstlisting}[caption={Condition estimation method}]
def estimate_system_conditions(self, state, raw_sensors, setpoint):
    """Estimate current operating conditions."""
    conditions = {}

    # Wind estimation (low-pass filtered)
    if 'wind' in raw_sensors:
        wind_sensor = raw_sensors['wind'].get('speed', 0.0)
        self.filtered_wind = (
            self.wind_filter_alpha * wind_sensor +
            (1 - self.wind_filter_alpha) * self.filtered_wind
        )
    conditions['wind_speed'] = self.filtered_wind

    # Tracking error
    position = state[:3]
    target = setpoint['position']
    tracking_error = np.linalg.norm(target - position)

    # Add distance penalty for large errors
    max_reasonable = 5.0
    if tracking_error > max_reasonable:
        tracking_error += (tracking_error - max_reasonable) * 0.3

    conditions['tracking_error'] = tracking_error

    # Estimation uncertainty (from EKF covariance)
    P = self.ekf.P
    conditions['estimation_uncertainty'] = np.sqrt(P[0,0] + P[1,1] + P[2,2])

    # Dynamic conditions
    conditions['angular_rates'] = np.linalg.norm(state[9:12])
    conditions['velocity'] = np.linalg.norm(state[3:6])
    conditions['altitude'] = -state[2]  # Convert NED to positive altitude

    return conditions
\end{lstlisting}

\section{Flight Mode Supervisor Integration}

The supervisor manages high-level flight modes and setpoint generation:

\begin{lstlisting}[caption={Flight mode supervisor update}]
def _supervisor_update(self, state, conditions, gps_ok, time):
    """Update supervisor and get setpoint/controller."""

    # Check PID contract assumptions
    pid_assumptions_ok = (
        conditions['wind_speed'] < 3.0 and
        conditions['tracking_error'] < 2.0 and
        conditions['estimation_uncertainty'] < 1.0 and
        conditions['angular_rates'] < 0.5 and
        gps_ok
    )

    # Update supervisor
    setpoint, controller_name = self.supervisor.update(
        state=state,
        conditions=conditions,
        pid_violated=not pid_assumptions_ok,
        gps_available=gps_ok,
        time=time
    )

    return setpoint, controller_name
\end{lstlisting}

class AdaptiveControlSystem:

        """Command landing sequence."""
        self.supervisor.command_land()

    def command_emergency(self):
        """Trigger emergency mode."""
        self.supervisor.command_emergency()

    def pre_flight_check(self, target):
        """Verify mission feasibility before takeoff."""
        # Check sensor contracts
        sensor_ok = self.contract_manager.check('sensor')

        # Verify target reachability
        mission_ok = self.planner.verify_mission([target])

        # Check vehicle state
        vehicle_ok = self._check_vehicle_ready()

        return {
            'sensors': sensor_ok,
            'mission': mission_ok,
            'vehicle': vehicle_ok,
            'ready': sensor_ok and mission_ok and vehicle_ok
        }

\section{Complete Control Step Implementation}

\begin{lstlisting}[caption={Complete control step method}]
def control_step(self, sensors, time=None):
    """Execute one control cycle."""
    if time is not None:
        self.time = time
    else:
        self.time += self.dt

    # 1. Process sensors
    measurements, raw_sensors = self._process_sensors(sensors)

    # 2. State estimation (CRITICAL: predict before update)
    self.ekf.predict(self.dt)
    state, estimation_ok = self.ekf.update(
        measurements, raw_sensors, self.time
    )
    self.current_state = state

    # Capture wind sensor
    if 'wind' in raw_sensors:
        self.last_wind_sensor = raw_sensors['wind'].get('speed', 0.0)

    # 3. Get current setpoint for condition estimation
    current_setpoint = self.supervisor.get_current_setpoint()

    # 4. Estimate conditions
    conditions = self.estimate_system_conditions(
        state, raw_sensors, current_setpoint
    )
    self.current_conditions = conditions

    # 5. Check GPS status
    sensor_contracts = self.ekf.check_sensor_contracts(raw_sensors)
    gps_ok = sensor_contracts['gps_valid']

    # 6. Horizon planning update (at 2 Hz)
    controller_rec, safety_margin = self.planner.update(
        state, conditions, self.time
    )

    # 7. Supervisor update
    setpoint, controller_name = self._supervisor_update(
        state, conditions, gps_ok, self.time
    )

    # 8. Controller switching
    if controller_name != self.controller_switcher.current_controller:
        self.controller_switcher.switch_to(controller_name)

    # 9. Compute nominal control
    nominal_control = self.controller_switcher.compute_control(
        state, setpoint
    )

    # 10. CBF safety filtering
    safe_control, cbf_intervened = self.cbf_filter.filter_control(
        state, nominal_control, self.time
    )

    # 11. Runtime monitoring
    self.runtime_monitor.update(
        state, safe_control, self.time, cbf_intervened
    )

    # 12. Telemetry logging
    self._log_telemetry(state, safe_control, conditions, self.time)

    return safe_control
\end{lstlisting}

\section{Error Handling}

The system implements graceful degradation:

\begin{lstlisting}[caption={Error handling strategy}]
def control_step_safe(self, sensors, time=None):
    """Control step with error handling."""
    try:
        return self.control_step(sensors, time)
    except EstimationError:
        # EKF failure - use last known state
        self.supervisor.command_emergency()
        return self._emergency_control()
    except SensorError:
        # Sensor failure - switch to robust mode
        self.controller_switcher.switch_to('Hinf')
        return self._degraded_control()
    except Exception as e:
        # Unknown error - safe landing
        logging.error(f"Control error: {e}")
        self.supervisor.command_land()
        return self._safe_landing_control()
\end{lstlisting}
